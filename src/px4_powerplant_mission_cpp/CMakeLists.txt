cmake_minimum_required(VERSION 3.16)
project(px4_powerplant_mission_cpp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(jsoncpp REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
set(ros_dependencies rclcpp px4_msgs nav_msgs geometry_msgs sensor_msgs std_msgs std_srvs visualization_msgs cv_bridge)
add_library(powerplant_core src/voxel_grid.cpp src/frontier_planner.cpp src/planning_index.cpp src/motion.cpp src/topology.cpp)
target_include_directories(powerplant_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(powerplant_core PUBLIC Eigen3::Eigen)
add_library(powerplant_mission src/map_snapshot.cpp src/offboard_node.cpp src/offboard_subscriptions_vehicle.cpp src/offboard_subscriptions_map.cpp src/offboard_subscriptions_camera.cpp src/offboard_control.cpp src/mission_state_machine.cpp src/mission_planning.cpp src/inspection_recorder.cpp src/inspection.cpp src/motion_runtime.cpp src/offboard_p2_motion.cpp src/offboard_p2_inspection.cpp src/offboard_p2_map.cpp)
ament_target_dependencies(powerplant_mission ${ros_dependencies})
target_compile_options(powerplant_mission PRIVATE -O1)
target_link_libraries(powerplant_mission powerplant_core JsonCpp::JsonCpp ${OpenCV_LIBRARIES} Threads::Threads)
add_executable(voxel_mapper_node src/voxel_mapper_node.cpp)
ament_target_dependencies(voxel_mapper_node ${ros_dependencies})
target_compile_options(voxel_mapper_node PRIVATE -O1)
target_link_libraries(voxel_mapper_node powerplant_core JsonCpp::JsonCpp ${OpenCV_LIBRARIES})
add_executable(offboard_mission_node src/offboard_main.cpp)
ament_target_dependencies(offboard_mission_node ${ros_dependencies})
target_link_libraries(offboard_mission_node powerplant_mission)
foreach(target powerplant_core powerplant_mission voxel_mapper_node offboard_mission_node)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS powerplant_core powerplant_mission voxel_mapper_node offboard_mission_node
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
install(FILES research_sources/sitl_body_envelope.json DESTINATION share/${PROJECT_NAME}/research_sources)
install(DIRECTORY config/ DESTINATION share/${PROJECT_NAME}/config)
